\begin{frame}[t]{O desafio de toda academia}
  \large
  \begin{itemize}

% 5. Gestão de uso e manutenção
% ---------------------------------------------------------
\begin{frame}[t]{Gestão de uso e manutenção preventiva: como é calculado}
  \small
  \begin{itemize}
    \item Cada ciclo \emph{ocupada $\rightarrow$ livre} é registrado como um \textbf{uso}, com horário e duração
    \item Por equipamento: tempo total de uso, número de usos e duração média
    \item \textbf{Mais e menos usadas:} ranking por tempo total e por frequência
    \item \textbf{Horários de pico:} uso agregado por hora do dia
    \item \textbf{Manutenção preventiva:} alerta quando as horas ou os usos acumulados atingem um limite configurável por equipamento
    \item \textbf{Estimativa de liberação:} duração média de uso e tempo desde que a máquina ficou ocupada
  \end{itemize}
\end{frame}

% 8. Nó: hardware e firmware
% ---------------------------------------------------------
\begin{frame}[t]{O nó sensor: hardware e firmware}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Hardware}
        \footnotesize
        \begin{itemize}
          \item \textbf{Blue Pill:} STM32F103C8T6, Cortex-M3 a 72~MHz
          \item \textbf{HC-SR04:} mede de 2~cm a 4~m; pulso de 10~$\mu$s no Trig e largura do pulso de Eco proporcional ao tempo de ida e volta
          \item \textbf{Módulo BLE serial} na UART
          \item ST-Link V2 para gravar; alimentação por USB no protótipo
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Firmware}
        \footnotesize
        \begin{itemize}
          \item Temporizador periódico dispara a medição; interrupção nas duas bordas e contador livre medem o Eco
          \item Distância = tempo $\times$ 343~m/s $\div$ 2
          \item Histerese, filtros de presença e de ausência (próximo slide)
          \item CPU em modo \emph{Sleep} entre as medições
          \item Quadro curto enviado pela UART ao módulo BLE
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% 10. Protocolo
% ---------------------------------------------------------
\begin{frame}[t]{Comunicação e protocolo}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \begin{block}{Quadro do nó}
        \footnotesize
        \texttt{N02,1,17}
        \begin{itemize}
          \item \texttt{02}: ID do nó
          \item \texttt{1}: ocupada (\texttt{0}: livre)
          \item \texttt{17}: número de sequência
          \item Repetido a cada 15~s como \textbf{sinal de vida}
          \item A sequência detecta perdas e duplicatas
          \item Sem sinal de vida por um intervalo configurável, o servidor marca o nó como \emph{offline}
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Rádio}
        \footnotesize
        \begin{itemize}
          \item \textbf{Protótipo:} o servidor conecta-se por BLE aos 2 ou 3 nós (biblioteca \emph{bleak})
          \item \textbf{Produto:} \emph{BLE advertising}, sem conexão, o que permite 100 nós sem esgotar conexões e com baixo consumo
          \item Quadro de produto inclui o nível de bateria
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% 11. Servidor, recomendação e site

\begin{frame}[t]{Servidor, recomendação e site}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \begin{block}{Servidor (Python, PC de mesa)}
        \footnotesize
        \begin{itemize}
          \item Recebe os quadros, decodifica e mantém o estado de cada máquina
          \item Acumula uso e frequência (slide 5)
          \item \textbf{Recomendação:} dada a rotina do frequentador (por exemplo, máquinas 12, 37, 5 e 18), devolve a primeira livre; se todas estiverem ocupadas, a que deve liberar primeiro
          \item Expõe uma API para o site
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Site (navegador)}
        \footnotesize
        \begin{itemize}
          \item \textbf{App do frequentador:} informa a rotina e vê a próxima máquina
          \item \textbf{Painel da gestão:} ocupação, uso e frequência por máquina, ranking, horários de pico e alertas
          \item Atualização em tempo real
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% ---------------------------------------------------------
% 12. Emulador
% ---------------------------------------------------------
\begin{frame}[t]{Emulador de até 100 nós}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{block}{Como funciona}
        \footnotesize
        \begin{itemize}
          \item Script que gera os \textbf{mesmos quadros} dos nós reais e os entrega ao servidor
          \item Os 2 ou 3 nós reais usam os IDs iniciais; o emulador cobre os demais
          \item Cada nó virtual segue um comportamento: livre, chega alguém, treina, descansa, sai
          \item Injeta perdas de quadro, atraso, nós mortos e passantes
          \item Modo \textbf{roteirizado} (cenários repetíveis) e \textbf{aleatório} (carga contínua)
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{block}{Como se valida}
        \footnotesize
        \begin{itemize}
          \item O emulador conhece o \textbf{estado verdadeiro} e compara com o que o servidor mostra
          \item Relógio acelerado para testar tempos de 90~s sem espera
          \item \textbf{Limite:} testa servidor e site, não o firmware nem o rádio real
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% 13. Simulação
% ---------------------------------------------------------
\begin{frame}[t]{Simulação e ferramentas}
  \begin{columns}[T]
    \begin{column}{0.35\textwidth}
      \begin{block}{Altium Designer}
        \footnotesize\raggedright
        \begin{itemize}
          \item Esquemático do nó
          \item Circuito: divisor de tensão do Eco (5~V para 3,3~V) e alimentação
          \item Eco imitado por fonte de pulso
          \item \textbf{Limite:} simulador SPICE, sem evidência de executar firmware
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{block}{Renode (principal)}
        \footnotesize\raggedright
        \begin{itemize}
          \item Código aberto; Linux, Windows e Mac
          \item Blue Pill F103: UART, GPIO, timers, I\textsuperscript{2}C, EXTI
          \item UART por socket TCP até o servidor
          \item Sem ADC e SPI (não usados)
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.30\textwidth}
      \begin{block}{Wokwi (alternativa)}
        \footnotesize\raggedright
        \begin{itemize}
          \item Blue Pill com o F103C8, o chip real
          \item HC-SR04 pronto (2 a 400~cm)
          \item Testes automatizados por cenários
          \item Sem DMA e RTC; ADC parcial
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  {\scriptsize O módulo BLE e o eco real não são simulados: viram serial ou socket e imitação. O Proteus (F103C6, só Windows) fica como opção. Teste de viabilidade na semana 1.\par}
\end{frame}

% 14. Premissas e limitações
% ---------------------------------------------------------
\begin{frame}[t]{Premissas do protótipo e limitações}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Premissas (ambiente ideal)}
        \footnotesize
        \begin{itemize}
          \item Sensor na posição correta, sem obstruções
          \item Cobertura BLE adequada
          \item Frequentadores com comportamento esperado
          \item Passantes tratados pelo filtro de presença
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Limitações}
        \footnotesize
        \begin{itemize}
          \item O ultrassom pode pegar quem passa; o filtro reduz, mas não elimina
          \item Conexão BLE limita os nós reais; 100 nós exigem \emph{advertising}
          \item Rádio e eco reais não são simulados
          \item Resultados valem para o ambiente ideal, não para uma academia real
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
